שהגמרא אומרת שצדיקים אחרי הסתלקותם הם למעלה יותר מכפי שהם בחיי חיותם בעולם זה, ובדמיוני הילדותי תיארתי לעצמי מה רם ונעלה הוא הסבא, ודמיוני הילדותי גדל בהתלהבותו.
בתוך כך שמעתי כמו אחד מדבר בלחש ובוכה. פתחתי את עיניי והנה אני רואה את הוד כ״ק אאמו״ר הרה״ק לבוש במעיל המשי ובשטריימיל של שבת, ענוד בחגורתו, עומד ליד השולחן הפינתי של הסבא — הוד כ״ק אאמו״ר הרה״ק מוהר״ש — מחזיק גליון נייר בידו, קורא ובוכה. נבהלתי מאד ויצאתי את החדר בלאט.
למדתי אז אצל המלמד ר׳ שמשון, אך בהיותו כעסן גדול, הייתי שואל כל דבר את מלמדי הקודם ר׳ יקותיאל, והוא הסביר לי אז את הענין בהתאם להבנתי. אז התעורר בי ההרגש, להתמסר להוד כ״ק אאמו״ר הרה״ק, לא רק כמו בן לאביו, אלא גם כמו חסיד לרבי.
ב. בליובאוויטש שמרו מאד על המנהגים, וביחוד על המנהגים שהיו מקובלים מהגאון הצדיק מהר״ל מפראג.
אחד המנהגים המקובלים הוא, שבמשך עשרת ימי תשובה עושים שלשה נרות, נר של תשובה בערב שבת שובה ושני נרות בערב יום כיפור, נר חי ונר נשמה. דייקו ליצור את הנרות משעוה, והיה משום כיבוד גדול לשזור את הנרות. אפילו הבאת המים לשריית השעוה וכן גם הדלקת העצים לחימום המים, היה בהם משום כיבוד רוחני.
בשעת הנחת הפתילות היו מכריזים על שמו של מי שעבורו שוזרים את החוט, כך קראו לזה. פעם אחת, בשנות ילדותי, כשהוד כ״ק אאמו״ר הרה״ק וכבוד אמי הרבנית היו בחוץ לארץ ואני הייתי בבית כבוד אמי זקנתי הרבנית רבקה, היה זה בין כסה לעשור, ובהיותי לבד בלי ההורים ורות עצב שרר עלי, בתוך כך שמעתי קוראים בשמי — ונחרדתי.
כבוד אמי זקנתי הרבנית קראה לי אליה ואמרה לי: אספר לך מה שחותני, ה״צמח צדק״, אמר לאביך בקטנותו.
הוד כ״ק אאמו״ר הרה״ק שאל את הסבא הוד כ״ק אאמו״ר הרה״ק ״צמח צדק״, מהי שבת תשובה? והוא ענה לו: תשובה שבתית. והפירוש הוא: בימות החול הרי לא נמצא בבית דבר איסור, ואילו בשבת נמצא בבית גם דבר איסור, אשר לכן כל מה שרוצים לעשות, יש לדעת בתחילה אם מותר לעשותו.
